%%                   tendril, sparkle, eight-ray star, pendant, ornament, monogram

%%   4. page plumbing (eso-pic, per-page seed)

%% ---------------------------------------------------------------- 1. palette
\definecolor{ink}{HTML}{1F3E8C}   % the only hue; everything is ink!NN
\definecolor{gold}{HTML}{C9A24C}  % single accent

%% ------------------------------------------------------------- 2. primitives
%% Every primitive is drawn at the origin of its own local frame and positioned
%% by the caller with a scope. Every primitive calls rand/rnd, so no two
%% instances are identical. Reminder: a coordinate component that contains
%% parentheses must be wrapped in braces, ({sin(\a)},{cos(\a)}).

%% --- petal: points +y, length 1.
%% Outline = egg-shaped parametric curve x(t),y(t) multiplied by a ruffle
%% rho(t) = a (1+sin t)/2 * sum_k c_k sin(f_k t + phi_k). The phases phi_k are
%% drawn once per petal, so rho is deterministic and the same outline can be
%% evaluated for the fill, the clip and the stroke. Layers:
%% gradient body, dark brush streaks, pale streaks near the base, dark veins,
%% inner rim (a wide stroke clipped to the inside), hairline outline.
\newcommand\petal{%
  \pgfmathsetmacro\pW{0.54+0.08*rand}%  half-width
  \pgfmathsetmacro\pR{0.028+0.018*rnd}% ruffle amplitude
  \pgfmathsetmacro\pPa{360*rnd}\pgfmathsetmacro\pPb{360*rnd}\pgfmathsetmacro\pPc{360*rnd}%
  \pgfmathsetmacro\pSk{0.05*rand}%      skew of the tip
  \def\petalpath{plot[smooth cycle,domain=-90:262,samples=44,variable=\t]
    ({\pW*cos(\t)*(1+0.35*sin(\t))*(1+\pR*(1+sin(\t))*(sin(5*\t+\pPa)+0.5*sin(11*\t+\pPb)+0.25*sin(23*\t+\pPc)))+\pSk*(1+sin(\t))/2},
     {0.5+0.5*sin(\t)*(1+\pR*(1+sin(\t))*(sin(5*\t+\pPa)+0.5*sin(11*\t+\pPb)+0.25*sin(23*\t+\pPc)))})}%
  \shade[save path=\pp,top color=ink!95!black,middle color=ink!72,bottom color=ink!22] \petalpath;
  \begin{scope}
    \clip[use path=\pp];
    \foreach \si in {1,...,9}{%
      \pgfmathsetmacro\sa{50*rand}\pgfmathsetmacro\sd{0.55+0.4*rnd}\pgfmathsetmacro\sw{0.015+0.025*rnd}%
      \fill[ink!90!black,opacity=0.22] (0,0.08)
        .. controls ({\sd*0.3*sin(\sa)-\sw},\sd*0.4) .. ({\sd*0.8*sin(\sa)},{\sd*cos(\sa)})
        .. controls ({\sd*0.3*sin(\sa)+\sw},\sd*0.4) .. cycle;}
    \foreach \si in {1,...,6}{%
      \pgfmathsetmacro\sa{45*rand}\pgfmathsetmacro\sd{0.25+0.35*rnd}\pgfmathsetmacro\sw{0.03+0.03*rnd}%
      \fill[white,opacity=0.35] (0,0.02)
        .. controls ({\sd*0.3*sin(\sa)-\sw},\sd*0.4) .. ({\sd*0.8*sin(\sa)},{\sd*cos(\sa)})
        .. controls ({\sd*0.3*sin(\sa)+\sw},\sd*0.4) .. cycle;}
    \foreach \si in {1,...,7}{%
      \pgfmathsetmacro\sa{48*rand}\pgfmathsetmacro\sd{0.7+0.28*rnd}%
      \draw[ink!85!black,opacity=0.5,line width=0.18pt] (0,0.05)
        .. controls ({\sd*0.3*sin(\sa)},\sd*0.4) .. ({\sd*0.85*sin(\sa)},{\sd*cos(\sa)});}
    \draw[use path=\pp,ink!85!black,opacity=0.4,line width=1.3pt];
  \end{scope}
  \draw[use path=\pp,ink!85!black,line width=0.3pt];
}

%% --- flower: five jittered petals, a pale fading heart, stamens. #1 = scale (petal length, cm)
\newcommand\flower[1]{%
  \begin{scope}[scale=#1,rotate=180*rand]
    \foreach \fk in {0,...,4}{%
      \pgfmathsetmacro\fa{\fk*72+7*rand}\pgfmathsetmacro\fl{0.93+0.1*rand}%
      \begin{scope}[rotate=\fa,scale=\fl]\petal\end{scope}}
    \fill[white,path fading=circle with fuzzy edge 20 percent,opacity=0.4] (0,0) circle (0.52);
    \fill[white,path fading=circle with fuzzy edge 20 percent,opacity=0.8] (0,0) circle (0.34);
    \foreach \fk in {1,...,16}{%
      \pgfmathsetmacro\fa{\fk*22.5+9*rand}\pgfmathsetmacro\fr{0.17+0.12*rnd}%
      \draw[ink!70,line width=0.22pt] (0,0.02) -- (\fa:\fr);
      \fill[gold!90!black] (\fa:\fr) circle (0.02);}
    \foreach \fk in {0,...,6}{\fill[gold!80!black] (\fk*51.4+20*rand:0.05) circle (0.016);}
    \fill[ink!30] (0,0) circle (0.02);
  \end{scope}}

%% --- leafshape: points +x. #1 length, #2 width ratio, #3 serration amplitude,
%% #4 number of teeth, #5 curl (tip offset as a fraction of length).
%% Upper edge and lower edge are two parametric plots joined at a pointed tip.
\newcommand\leafshape[5]{%
  \pgfmathsetmacro\lL{#1}\pgfmathsetmacro\lW{#2}\pgfmathsetmacro\lS{#3}\pgfmathsetmacro\lN{#4}\pgfmathsetmacro\lC{#5}%
  \pgfmathsetmacro\lPh{180*rnd}%
  \def\leafpath{%
    plot[smooth,domain=0:180,samples=36,variable=\t]
      ({\lL*(1-cos(\t))/2},{\lW*\lL*sin(\t)*(1+\lS*(1-abs(sin(\lN*\t+\lPh))))+\lC*\lL*((1-cos(\t))/2)^2})
    plot[smooth,domain=180:0,samples=36,variable=\t]
      ({\lL*(1-cos(\t))/2},{-\lW*\lL*sin(\t)*(1+\lS*(1-abs(sin(\lN*\t+\lPh+40))))+\lC*\lL*((1-cos(\t))/2)^2})
    -- cycle}%
  \shade[save path=\lp,left color=ink!92!black,middle color=ink!70,right color=ink!32] \leafpath;
  \begin{scope}
    \clip[use path=\lp];
    \draw[white,opacity=0.6,line width=0.3pt] (0,0) .. controls (0.5*\lL,0) .. (0.98*\lL,\lC*\lL);
    \foreach \lt in {0.12,0.24,...,0.84}{%
      \pgfmathsetmacro\lv{(1-\lt)*1.3*\lW*\lL}\pgfmathsetmacro\lx{\lt*\lL}\pgfmathsetmacro\ly{\lC*\lL*\lt*\lt}%
      \draw[white,opacity=0.45,line width=0.18pt]
        (\lx,\ly) .. controls (\lx+0.3*\lv,\ly+0.5*\lv) .. (\lx+0.55*\lv,\ly+\lv)
        (\lx,\ly) .. controls (\lx+0.3*\lv,\ly-0.5*\lv) .. (\lx+0.55*\lv,\ly-\lv);}
    \foreach \si in {1,2,3}{%
      \pgfmathsetmacro\sx{\lL*(0.2+0.6*rnd)}\pgfmathsetmacro\sy{\lW*\lL*0.6*rand}%
      \fill[ink!90!black,opacity=0.18] (0.05*\lL,0)
        .. controls (0.5*\sx,\sy*0.6+0.02*\lL) .. (\sx,\sy) .. controls (0.5*\sx,\sy*0.6-0.02*\lL) .. cycle;}
  \end{scope}
  \draw[use path=\lp,ink!85!black,line width=0.26pt];
}
\newcommand\serleaf[1]{\leafshape{#1}{0.26+0.04*rand}{0.10}{6}{0.14*rand}}  % serrated leaflet
\newcommand\blade[2]{\leafshape{#1}{0.07+0.02*rand}{0}{1}{#2}}              % grass blade, #2 = curl

%% --- dissected leaf: a rib with three pairs of serrated lobes and a terminal lobe
\newcommand\dissected[1]{%
  \pgfmathsetmacro\dL{#1}%
  \draw[ink!85!black,line width=0.4pt] (0,0) .. controls (0.5*\dL,0.02*\dL) .. (0.78*\dL,0.06*\dL);
  \foreach \dp/\ds in {0.66/0.32,0.46/0.40,0.22/0.46}{%
    \begin{scope}[shift={(\dp*\dL,0.03*\dL)},rotate=52+10*rand]\serleaf{\ds*\dL}\end{scope}
    \begin{scope}[shift={(\dp*\dL,0.03*\dL)},rotate=-52+10*rand]\serleaf{\ds*\dL}\end{scope}}
  \begin{scope}[shift={(0.76*\dL,0.06*\dL)},rotate=5*rand]\serleaf{0.34*\dL}\end{scope}}

%% --- simple leaf (cheap; used for fern leaflets)
\newcommand\leaf[1]{%
  \pgfmathsetmacro\lfL{#1}\pgfmathsetmacro\lfW{(0.30+0.06*rand)*\lfL}\pgfmathsetmacro\lfC{0.10*rand*\lfL}%
  \def\leafoutline{(0,0) .. controls (0.25*\lfL,\lfW) and (0.80*\lfL,0.85*\lfW+\lfC) .. (\lfL,\lfC)
          .. controls (0.80*\lfL,-0.85*\lfW+\lfC) and (0.25*\lfL,-\lfW) .. cycle}%
  \shade[left color=ink!92!black,middle color=ink!70,right color=ink!30] \leafoutline;
  \begin{scope}\clip \leafoutline;
    \draw[white,opacity=0.55,line width=0.3pt] (0,0) .. controls (0.5*\lfL,0.3*\lfC) .. (0.97*\lfL,\lfC);
    \foreach \lt in {0.2,0.4,0.6,0.8}{%
      \pgfmathsetmacro\lv{(1-\lt)*1.2*\lfW}\pgfmathsetmacro\lx{\lt*\lfL}\pgfmathsetmacro\ly{\lt*\lfC}%
      \draw[white,opacity=0.4,line width=0.2pt]
        (\lx,\ly) .. controls (\lx+0.25*\lv,\ly+0.45*\lv) .. (\lx+0.45*\lv,\ly+\lv)
        (\lx,\ly) .. controls (\lx+0.25*\lv,\ly-0.45*\lv) .. (\lx+0.45*\lv,\ly-\lv);}
  \end{scope}
  \draw[ink!85!black,line width=0.3pt] \leafoutline;}

%% --- fern: curved rib, leaflets placed by markings and tapered by their sequence number
\newcommand\fern[1]{%
  \pgfmathsetmacro\fnL{#1}%
  \draw[ink!85!black,line width=0.45pt,
        postaction={decorate,decoration={markings,
          mark=between positions 0.06 and 0.94 step 0.055 with {%
            \pgfmathtruncatemacro\mk{\pgfkeysvalueof{/pgf/decoration/mark info/sequence number}}%
            \pgfmathsetmacro\ms{\fnL*0.24*(1-0.05*\mk)}%
            \begin{scope}[rotate=58+8*rand]\leaf{\ms}\end{scope}
            \begin{scope}[rotate=-58+8*rand]\leaf{\ms}\end{scope}}}}]
    (0,0) .. controls (0.4*\fnL,0.05*\fnL) and (0.75*\fnL,0.25*\fnL) .. (\fnL,0.45*\fnL);}

%% --- floret: a five-petal flower a few millimetres across
\newcommand\floret{%
  \pgfmathsetmacro\fph{72*rnd}%
  \foreach \fk in {0,...,4}{\fill[ink!45,draw=ink!85!black,line width=0.15pt] (\fk*72+\fph:0.046) circle (0.03);}
  \fill[ink!90!black] (0,0) circle (0.016);}

%% --- spray: recursive ternary branching; florets and stray dots at the leaves.
%% #1 angle, #2 length, #3 depth
\newcommand\spray[3]{%
  \ifnum#3>0
    \pgfmathsetmacro\sb{12*rand}%
    \draw[ink!80,line width=0.28pt] (0,0) to[bend left=\sb] (#1:#2);
    \begin{scope}[shift={(#1:#2)}]
      \pgfmathtruncatemacro\sd{#3-1}%
      \foreach \sa in {-40,0,40}{%
        \pgfmathsetmacro\sna{#1+\sa+14*rand}\pgfmathsetmacro\snl{#2*(0.6+0.12*rand)}%
        \spray{\sna}{\snl}{\sd}}
    \end{scope}
  \else
    \floret
    \foreach \fk in {1,2}{\fill[ink!70] (360*rnd:0.1+0.08*rnd) circle (0.009);}
  \fi}

%% --- arch spray: one arching stalk with small sprays along it. #1 length
\newcommand\archspray[1]{%
  \pgfmathsetmacro\asL{#1}%
  \draw[ink!85!black,line width=0.45pt,
    postaction={decorate,decoration={markings,
      mark=between positions 0.2 and 0.96 step 0.085 with {%
        \pgfmathtruncatemacro\mk{\pgfkeysvalueof{/pgf/decoration/mark info/sequence number}}%
        \pgfmathsetmacro\mside{ifthenelse(mod(\mk,2)==0,1,-1)}%
        \pgfmathsetmacro\msa{\mside*(55+20*rand)}\pgfmathsetmacro\msl{0.26+0.08*rand}%
        \spray{\msa}{\msl}{2}}}}]
    (0,0) .. controls (0.25*\asL,0.5*\asL) and (0.75*\asL,0.6*\asL) .. (\asL,0.25*\asL);
  \begin{scope}[shift={(\asL,0.25*\asL)}]\floret\end{scope}}

%% --- umbel (Queen Anne's lace): rays fanning upward, each ending in a small secondary umbel
\newcommand\umbel[1]{%
  \pgfmathsetmacro\uS{#1}%
  \foreach \ur in {1,...,11}{%
    \pgfmathsetmacro\ua{25+\ur*11+6*rand}\pgfmathsetmacro\ul{\uS*(0.78+0.22*rnd)}%
    \draw[ink!80,line width=0.26pt] (0,0) .. controls (\ua+5:0.5*\ul) .. (\ua:\ul);
    \begin{scope}[shift={(\ua:\ul)}]
      \foreach \us in {1,...,6}{%
        \pgfmathsetmacro\ub{\ua-52+\us*17+5*rand}\pgfmathsetmacro\ubl{\uS*(0.15+0.07*rnd)}%
        \draw[ink!80,line width=0.18pt] (0,0) -- (\ub:\ubl);
        \fill[ink!55,draw=ink!85!black,line width=0.12pt] (\ub:\ubl) circle (0.021);}
      \fill[ink!85] (0,0) circle (0.013);
    \end{scope}}
  \foreach \ub in {-30,-10,10,30}{\draw[ink!75,line width=0.25pt] (0,0) .. controls (\ub+90:0.16*\uS) .. (\ub+90+12:0.28*\uS);}}

%% --- tendril: logarithmic spiral. #1 size
\newcommand\tendril[1]{%

%% --- sparkles
\newcommand\sparklepath[1]{% four-point star with concave sides
    (0,#1) .. controls (0.07*#1,0.07*#1) and (0.07*#1,0.07*#1) .. (#1,0)
           .. controls (0.07*#1,-0.07*#1) and (0.07*#1,-0.07*#1) .. (0,-#1)
           .. controls (-0.07*#1,-0.07*#1) and (-0.07*#1,-0.07*#1) .. (-#1,0)
           .. controls (-0.07*#1,0.07*#1) and (-0.07*#1,0.07*#1) .. cycle}
\newcommand\sparkle[2]{% #1 radius, #2 colour
  \fill[#2] \sparklepath{#1};
  \begin{scope}[rotate=45,scale=0.45]\fill[#2] \sparklepath{#1};\end{scope}}
\newcommand\starpath[1]{% eight rays, the vertical pair longer
  (90:1.3*#1) .. controls (67.5:0.07*#1) and (67.5:0.07*#1) .. (45:0.45*#1)
  .. controls (22.5:0.07*#1) and (22.5:0.07*#1) .. (0:#1)
  .. controls (-22.5:0.07*#1) and (-22.5:0.07*#1) .. (-45:0.45*#1)
  .. controls (-67.5:0.07*#1) and (-67.5:0.07*#1) .. (-90:1.3*#1)
  .. controls (-112.5:0.07*#1) and (-112.5:0.07*#1) .. (-135:0.45*#1)
  .. controls (-157.5:0.07*#1) and (-157.5:0.07*#1) .. (180:#1)
  .. controls (157.5:0.07*#1) and (157.5:0.07*#1) .. (135:0.45*#1)
  .. controls (112.5:0.07*#1) and (112.5:0.07*#1) .. cycle}
\newcommand\bigsparkle[1]{% gold core fading to ink at the tips, rays and dots around it
  \shade[inner color=gold!85!white,outer color=ink!85!black] \starpath{#1};
  \draw[ink!85!black,line width=0.22pt] \starpath{#1};
  \foreach \ha in {0,45,...,315}{\draw[ink!70,line width=0.22pt] (\ha:1.5*#1) -- (\ha:1.75*#1);\fill[ink!70] (\ha:1.85*#1) circle (0.02);}
  \foreach \ha in {22.5,67.5,...,337.5}{\fill[ink!60] (\ha:1.25*#1) circle (0.014);}
  \foreach \ha in {0,90,180,270}{\fill[gold] (\ha:2.15*#1) circle (0.025);}}

%% --- pendant: a sparkle, a dotted chain, a diamond charm, a gold bead. #1 length
\newcommand\pendant[1]{%

  \fill[ink!85] (0,-#1) -- (0.07,-#1-0.13) -- (0,-#1-0.26) -- (-0.07,-#1-0.13) -- cycle;
  \draw[ink!70,line width=0.3pt] (0,-#1-0.26) -- (0,-#1-0.36);
  \fill[gold] (0,-#1-0.42) circle (0.04);}

%% --- ornament: one of five small marks, chosen at random
\newcommand\ornament{%
  \pgfmathtruncatemacro\ok{random(0,4)}%
  \ifcase\ok
    \sparkle{0.11}{gold}%
  \or \draw[ink!80,line width=0.28pt] (0,0) circle (0.075) (-0.075,0) -- (0.075,0) (0,-0.075) -- (0,0.075);%
  \or \fill[ink!80] (0,0.11) -- (0.055,0) -- (0,-0.11) -- (-0.055,0) -- cycle;%
  \or \fill[ink!75] (0,0.12) .. controls (0.07,0) and (0.07,-0.08) .. (0,-0.12) .. controls (-0.07,-0.08) and (-0.07,0) .. cycle;%
  \or \foreach \oa in {0,60,...,300}{\fill[ink!70] (\oa:0.065) circle (0.011);}\fill[gold] (0,0) circle (0.018);%
  \fi}

\subsection{The garden in the margin}

Everything in the left margin of this page is drawn by \TeX\ at shipout time from a few
hundred lines of TikZ. There are no image files, no imported paths and no external
programs; every petal is a parametric curve evaluated while the page is being written.
This section records how it is done, so that the margin documents itself.

\begin{definition}
A \emph{primitive} is a closed outline $\gamma\colon[0,1]\to\mathbb{R}^2$ authored by hand
together with a small set of free parameters $(p_1,\dots,p_k)$. An \emph{instance} of the
primitive samples the parameters from fixed intervals and draws $\gamma$ in a local frame
chosen by the caller. The composition is \emph{authored}; only the parameters are
\emph{sampled}. This is what ``procedural'' means here: no flower is grown from a grammar,
but no two petals share a shape.
\end{definition}

\begin{construction}[Petal]
A petal of length $1$ pointing along $+y$ is the egg curve
\begin{align*}
  x(\theta)&=w\cos\theta\,(1+0.35\sin\theta)\,\bigl(1+\rho(\theta)\bigr),\\
  y(\theta)&=\tfrac12+\tfrac12\sin\theta\,\bigl(1+\rho(\theta)\bigr),
  \qquad \theta\in[-90^\circ,270^\circ),
\end{align*}
with half-width $w\in[0.46,0.62]$ and a \emph{ruffle}
\begin{align*}
  \rho(\theta)&=a\,\frac{1+\sin\theta}{2}\,\bigl(\sin(5\theta+\varphi_1)
  +\tfrac12\sin(11\theta+\varphi_2)+\tfrac14\sin(23\theta+\varphi_3)\bigr),\\
  a&\in[0.028,0.046].
\end{align*}
The factor $(1+\sin\theta)/2$ vanishes at the base and is largest at the tip, so the petal
attaches cleanly and frays where petals fray. The phases $\varphi_k$ are drawn once per
petal; after that $\rho$ is deterministic, which matters because the same outline is
needed three times: once as a gradient fill, once as a clipping path and once as a
hairline stroke. The fill is an axis gradient from \texttt{ink!22} at the base to
\texttt{ink!95!black} at the tip. Inside the clip, nine dark and six pale lens-shaped
streaks radiate from the base at random angles, seven hairline veins run to the edge, and
the outline is stroked again $1.3\,$pt wide so that only its inner half shows, giving the
dark rim. The brushed texture of the petal is nothing more than those fifteen translucent
lenses.
\end{construction}

\begin{construction}[Flower]
Five petals are placed at $72^\circ$ intervals with $\pm7^\circ$ of jitter and lengths in
$[0.83,1.03]$; the whole is rotated by a random angle. A white disc with a fuzzy-edged
\emph{fading} is laid over the centre, which is what pulls the petal bases toward white
without any seam. Sixteen stamens of random length end in gold dots; seven more gold
dots sit at the heart.
\end{construction}

\begin{construction}[Leaves]
A leaf of length $L$ along $+x$ has upper and lower edges
\begin{align*}
  x(t)&=L\,\frac{1-\cos t}{2},\\
  y_\pm(t)&=\pm\,\lambda L\sin t\,\bigl(1+s\,(1-\lvert\sin(n t+\psi)\rvert)\bigr)
           +c\,L\Bigl(\tfrac{1-\cos t}{2}\Bigr)^{2},
\end{align*}
for $t\in[0^\circ,180^\circ]$. The width ratio $\lambda$, serration depth $s$, tooth count
$n$ and curl $c$ distinguish the species: $(\lambda,s,n)=(0.26,0.10,6)$ gives a serrated
leaflet, $(0.08,0,1)$ with $c\in[-0.35,0.35]$ a grass blade. The two edges are separate
smooth plots joined at the tip, so the tip is pointed. A \emph{dissected} leaf is a rib
carrying three pairs of serrated leaflets of decreasing size at $\pm52^\circ$ and a terminal
leaflet; a \emph{fern} is a rib with leaflets placed by path markings and scaled by their
sequence number so that they taper.
\end{construction}

\begin{construction}[Sprays and umbels]
A \emph{spray} is a ternary recursion of depth $d$: each segment forks into three
segments at $-40^\circ,0^\circ,+40^\circ$ ($\pm14^\circ$) of $0.6$ its length ($\pm12\%$),
and at depth $0$ draws a five-petal floret two millimetres across and a couple of stray
dots. An \emph{arch spray} is one long bent stalk with depth-2 sprays hung alternately
left and right along it. An \emph{umbel} fans eleven rays upward, each ending in a
secondary umbel of six short rays with a dot at every tip; four short bracts curl up from
the base. Tendrils are logarithmic spirals $r=0.06\,e^{0.0036\,\theta}$.
\end{construction}

\subsubsection{Composition}

Two stalks run the height of the page: a thick vine and a thinner companion that crosses
it four times. Each is a smooth plot through eleven control points whose $x$ coordinates
carry $\pm3\,$mm of jitter. Leaves are not placed by hand; they are hung along the stalks
by \texttt{decorations.markings}, which executes arbitrary drawing code at fractions of
the arc length in a frame aligned with the tangent, so every leaf follows the direction of
its stem. The same mechanism drops named coordinates at six fractions of the vine, and the
larger elements --- flowers, bells, dissected leaves, the arch sprays --- are attached to
short side branches grown from those anchors. Elements are drawn in depth order: sprays
and ferns first, foliage, then bells and buds, then flowers, then the scattered ornaments.
Everything is clipped to the margin column.

\begin{remark}[Procedural is not random]
The seed of the generator is set to a function of the page number before the garden is
drawn. Page $n$ therefore receives the same garden on every compilation, and a different
one from page $n+1$. The variation the reader notices between pages is real; the
reproducibility is what makes the document stable enough to typeset.
\end{remark}

\begin{proposition}
Let $P$ be the number of primitive instances on a page and $S$ the mean number of samples
per outline. The cost of a page is $\Theta(PS)$ evaluations of \texttt{pgfmath}, and in
practice on the order of a minute per page for $P\approx 400$ and $S\approx 50$ under
\texttt{pdflatex}.
\end{proposition}

\begin{proof}[Proof sketch]
Each outline is evaluated once for its fill, and its clip and stroke reuse the saved soft
path (\texttt{save path}/\texttt{use path}), so the per-primitive cost is the plot itself;
the dozen or so textural strokes inside the clip are constant work. The constant is
dominated by \texttt{pgfmath}, which computes in \TeX\ dimensions: integers and
intermediate results must stay below $16\,384$, which is also why the page seed is
$97\,n+1030$ and not something larger.
\end{proof}

\begin{exercise}
The composition above is authored. Replace it by a growth process in the spirit of Wong,
Zongker and Salesin (\emph{Computer-generated floral ornament}, SIGGRAPH 1998): starting
from one seed on the vine, repeatedly choose a primitive, find the largest free disc in
the margin region that touches an existing element, and place the primitive there if it
fits, until no disc larger than a threshold remains. The ornament then adapts to any text
block, and a page with a wide display equation grows a wider garden. Pure
\texttt{pgfmath} is a poor host for the bookkeeping; \texttt{luatex} with the layout
computed in Lua and the drawing left to the primitives above is the natural
implementation.
\end{exercise}

\subsubsection{Building and sharing}

The whole document is one file. It compiles with \texttt{pdflatex} in about half a
minute per page, and with \texttt{lualatex} or \texttt{xelatex} unchanged. To share it,
the three routes that work are, in increasing order of effort: paste the file into a blank
Overleaf project and share the link; put it in a Git repository with a continuous
integration job that compiles the PDF on every push and attaches it to tagged releases;
or move the primitives of Section~3.2 into a package file so that any document can
\texttt{\textbackslash usepackage} the garden and choose its own seed, column and hue.

\begin{remark}
Two mistakes cost an afternoon each and are worth recording. A \texttt{\textbackslash
foreach} variable must not be an existing control sequence; \texttt{\textbackslash fi}
in particular is a \TeX\ primitive and silently destroys the conditional that contains it.
And a coordinate whose component contains parentheses, such as $(\sin\alpha,\cos\alpha)$,
must have each component wrapped in braces, because TikZ reads a coordinate up to the
first closing parenthesis it meets.
\end{remark}
